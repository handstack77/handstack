SELECT TL."LogNo"
    , TL."ServerID"
    , TL."RunningEnvironment"
    , TL."GlobalID"
    , TL."ProgramName"
    , TL."ApplicationID"
    , TL."ProjectID"
    , TL."TransactionID"
    , TL."ServiceID"
    , TL."Acknowledge" 
    , TL."Type"
    , TL."Flow"
    , TL."Level"
    , TL."Format"
    , TL."UserID"
    , TO_CHAR(TL."CreatedAt", 'YYYY-MM-DD HH24:MI:SS') AS "CreatedAt"
	, TL."IpAddress"
	, TL."DeviceID"
	, TL."ProgramID"
FROM 
    "{TableName}" TL
WHERE (:GlobalID IS NULL OR TL."GlobalID" = :GlobalID)
    AND (:Environment IS NULL OR TL."RunningEnvironment" = :Environment)
    AND (:ProjectID IS NULL OR TL."ProjectID" = :ProjectID)
    AND (:ServerID IS NULL OR TL."ServerID" = :ServerID)
    AND (:TransactionID IS NULL OR TL."TransactionID" = :TransactionID)
    AND (:ServiceID IS NULL OR TL."ServiceID" = :ServiceID)
    AND TL."CreatedAt" BETWEEN TO_TIMESTAMP(SUBSTR(:StartedAt, 1, 19), 'YYYY-MM-DD HH24:MI:SS') AND TO_TIMESTAMP(SUBSTR(:EndedAt, 1, 19), 'YYYY-MM-DD HH24:MI:SS')
ORDER BY TL."LogNo" DESC
FETCH FIRST 500 ROWS ONLY
